## Example Implementation

```typescript
import { z } from "zod";
import { Entity, type EntityData } from "./base/entity.ts";

export interface CardAccountData extends EntityData {
  name: string;
  issuer: string;
  last4: string;
  billingCycle: number;
  creditLimitCents?: number;
  ownerId: string;
  isActive: boolean;
}

export class CardAccount extends Entity {
  private _name: string;
  private _issuer: string;
  private _last4: string;
  private _billingCycle: number;
  private _creditLimitCents?: number;
  private _ownerId: string;

  constructor(data: CardAccountData) {
    super(data);
    const validated = CardAccount.schema.parse(data);
    this._name = validated.name.trim();
    this._issuer = validated.issuer.trim();
    this._last4 = validated.last4;
    this._billingCycle = validated.billingCycle;
    this._creditLimitCents = validated.creditLimitCents;
    this._ownerId = validated.ownerId;
  }

  get name(): string {
    return this._name;
  }
  get issuer(): string {
    return this._issuer;
  }
  get last4(): string {
    return this._last4;
  }
  get billingCycle(): number {
    return this._billingCycle;
  }
  get creditLimitCents(): number | undefined {
    return this._creditLimitCents;
  }
  get ownerId(): string {
    return this._ownerId;
  }

  canDelete(): boolean {
    // Placeholder - actual check in service layer (needs transaction repository)
    return true;
  }

  archive(): void {
    this.toggleActive();
  }

  override toJSON(): CardAccountData {
    return {
      ...super.toJSON(),
      name: this._name,
      issuer: this._issuer,
      last4: this._last4,
      billingCycle: this._billingCycle,
      creditLimitCents: this._creditLimitCents,
      ownerId: this._ownerId,
    };
  }

  static parse(data: unknown): CardAccount {
    const validated = this.bodySchema.parse(data);
    return new CardAccount(validated);
  }

  static override get schema() {
    return z.object({
      name: z.string().min(1).max(255).trim(),
      issuer: z.string().min(1).max(255).trim(),
      last4: z
        .string()
        .length(4)
        .regex(/^\d{4}$/, "Must be 4 digits"),
      billingCycle: z.number().int().min(1).max(31),
      creditLimitCents: z.number().int().positive().optional(),
      ownerId: z.string().uuid(),
    });
  }

  static readonly createSchema = this.schema;

  static override get bodySchema() {
    return super.bodySchema.extend(this.schema.shape);
  }
}
```

## Example Test

```typescript
import { assertEquals, assertThrows } from "@std/assert";
import { CardAccount } from "@shared/entities/cardAccount.ts";

Deno.test("CardAccount - creates valid instance", () => {
  const data = {
    name: "Chase Sapphire",
    issuer: "Chase",
    last4: "1234",
    billingCycle: 15,
    ownerId: crypto.randomUUID(),
    isActive: true,
  };

  const cardAccount = new CardAccount(data);

  assertEquals(cardAccount.name, "Chase Sapphire");
  assertEquals(cardAccount.last4, "1234");
  assertEquals(cardAccount.billingCycle, 15);
});

Deno.test("CardAccount - validates last4 must be 4 digits", () => {
  const data = {
    name: "Test",
    issuer: "Test",
    last4: "12", // Invalid - not 4 digits
    billingCycle: 15,
    ownerId: crypto.randomUUID(),
    isActive: true,
  };

  assertThrows(() => new CardAccount(data));
});

Deno.test("CardAccount - validates billingCycle range", () => {
  const baseData = {
    name: "Test",
    issuer: "Test",
    last4: "1234",
    ownerId: crypto.randomUUID(),
    isActive: true,
  };

  // Valid edge cases
  const valid1 = new CardAccount({ ...baseData, billingCycle: 1 });
  assertEquals(valid1.billingCycle, 1);

  const valid31 = new CardAccount({ ...baseData, billingCycle: 31 });
  assertEquals(valid31.billingCycle, 31);

  // Invalid cases
  assertThrows(() => new CardAccount({ ...baseData, billingCycle: 0 }));
  assertThrows(() => new CardAccount({ ...baseData, billingCycle: 32 }));
});
```
