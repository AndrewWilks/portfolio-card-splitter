## Goal

Create comprehensive integration tests that verify the complete CardAccount, CardAccountSettings, Card, and updated Transaction backend functionality works end-to-end with the database.

## Context

Integration tests differ from unit tests:

- Unit tests: Instantiation and method existence only (no database)
- Integration tests: Full API flows with database operations
- Integration tests verify repository → service → route → database interactions
- Integration tests use test database and clean up after themselves

The project doesn't currently have a formal integration test structure, so we'll follow common patterns:

- Separate integration test directory: `backend/__tests__/integration/`
- Test setup/teardown utilities
- Full API request/response cycle testing
- Database state verification

### Example Integration Test: CardAccount

**File**: `backend/__tests__/integration/cardAccount.test.ts`

```typescript
import { assert, assertEquals, assertExists } from "@std/assert";
import { CardAccountService } from "@backend/services";
import {
  CardAccountRepository,
  CardAccountSettingsRepository,
} from "@backend/repositories";
import { cleanDatabase, createTestUser } from "./testHelpers.ts";
import { createCardAccountData } from "./factories.ts";

Deno.test(
  "CardAccount Integration - Create CardAccount with Settings",
  async () => {
    await cleanDatabase();

    const user = await createTestUser();
    const service = new CardAccountService(
      new CardAccountRepository(),
      new CardAccountSettingsRepository()
    );

    const data = createCardAccountData(user.id);
    const result = await service.createCardAccount(data, user.id);

    // Verify CardAccount created
    assertExists(result.cardAccount);
    assertEquals(result.cardAccount.name, data.name);
    assertEquals(result.cardAccount.ownerId, user.id);

    // Verify Settings created
    assertExists(result.settings);
    assertEquals(result.settings.cardAccountId, result.cardAccount.id);
    assert(result.settings.hasInterestFreePeriod);

    await cleanDatabase();
  }
);

Deno.test("CardAccount Integration - List by Owner", async () => {
  await cleanDatabase();

  const user1 = await createTestUser();
  const user2 = await createTestUser();

  const service = new CardAccountService(
    new CardAccountRepository(),
    new CardAccountSettingsRepository()
  );

  // Create accounts for both users
  await service.createCardAccount(createCardAccountData(user1.id), user1.id);
  await service.createCardAccount(createCardAccountData(user1.id), user1.id);
  await service.createCardAccount(createCardAccountData(user2.id), user2.id);

  // List user1's accounts
  const user1Accounts = await service.listCardAccounts(user1.id);
  assertEquals(user1Accounts.length, 2);

  // List user2's accounts
  const user2Accounts = await service.listCardAccounts(user2.id);
  assertEquals(user2Accounts.length, 1);

  await cleanDatabase();
});

Deno.test(
  "CardAccount Integration - Soft Delete Prevention with Transactions",
  async () => {

    const member = await createTestMember(user.id);
```

### Example Integration Test: Transaction with CardAccount

**File**: `backend/__tests__/integration/transaction.test.ts`

```typescript
import { assert, assertEquals, assertRejects } from "@std/assert";
import { TransactionService } from "@backend/services";
import {
  TransactionRepository,
  MerchantRepository,
  TagRepository,
  MemberRepository,
  PotRepository,
  CardAccountRepository,
  CardRepository,
} from "@backend/repositories";
import {
  cleanDatabase,
  createTestUser,
  createTestMerchant,
  createTestMember,
} from "./testHelpers.ts";
import { createCardAccountData, createTransactionData } from "./factories.ts";

Deno.test("Transaction Integration - Create with CardAccount", async () => {
  await cleanDatabase();

  const user = await createTestUser();
  const merchant = await createTestMerchant(user.id);
  const member = await createTestMember(user.id);

  const cardAccountService = new CardAccountService(
    new CardAccountRepository(),
    new CardAccountSettingsRepository()
  );

  const transactionService = new TransactionService(
    new TransactionRepository(),
    new MerchantRepository(),
    new TagRepository(),
    new MemberRepository(),
    new PotRepository(),
    new CardAccountRepository(),
    new CardRepository()
  );

  // Create CardAccount
  const { cardAccount } = await cardAccountService.createCardAccount(
    createCardAccountData(user.id),
    user.id
  );

  // Create Transaction
  const txData = createTransactionData(cardAccount.id, member.id, merchant.id);

  const transaction = await transactionService.createTransaction({
    ...txData,
    transactionDate: txData.transactionDate.toISOString(),
  });

  assertEquals(transaction.cardAccountId, cardAccount.id);
  assertEquals(transaction.memberId, member.id);
  assertEquals(transaction.merchantId, merchant.id);

  await cleanDatabase();
});
```
