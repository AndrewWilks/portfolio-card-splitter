## Goal

Create Card entity representing physical/virtual card numbers within a CardAccount (for member attribution and filtering).

## Context

Cards are **optional** entities that help track "who swiped" when multiple people have cards on the same CardAccount. They enable:

- Linking a Card to a Member for attribution
- Filtering transactions by specific Card
- Supporting multiple cards per CardAccount
- Tracking card nicknames and last4 digits

**Important**: Cards do NOT affect splits or payments - they're purely for attribution and filtering.

## Example Implementation

```typescript
import { z } from "zod";
import { Entity, type EntityData } from "./base/entity.ts";

export interface CardData extends EntityData {
  cardAccountId: string;
  memberId?: string;
  nickname?: string;
  last4?: string;
  isActive: boolean;
}

export class Card extends Entity {
  private _cardAccountId: string;
  private _memberId?: string;
  private _nickname?: string;
  private _last4?: string;

  constructor(data: CardData) {
    super(data);
    const validated = Card.schema.parse(data);
    this._cardAccountId = validated.cardAccountId;
    this._memberId = validated.memberId;
    this._nickname = validated.nickname?.trim();
    this._last4 = validated.last4;
  }

  get cardAccountId(): string {
    return this._cardAccountId;
  }
  get memberId(): string | undefined {
    return this._memberId;
  }
  get nickname(): string | undefined {
    return this._nickname;
  }
  get last4(): string | undefined {
    return this._last4;
  }

  get displayName(): string {
    if (this._nickname) {
      return this._last4
        ? `${this._nickname} (••${this._last4})`
        : this._nickname;
    }
    return this._last4 ? `Card ••${this._last4}` : "Card";
  }

  canDelete(): boolean {
    // Placeholder - actual check in service layer (needs transaction repository)
    return true;
  }

  archive(): void {
    this.toggleActive();
  }

  override toJSON(): CardData {
    return {
      ...super.toJSON(),
      cardAccountId: this._cardAccountId,
      memberId: this._memberId,
      nickname: this._nickname,
      last4: this._last4,
    };
  }

  static parse(data: unknown): Card {
    const validated = this.bodySchema.parse(data);
    return new Card(validated);
  }

  static override get schema() {
    return z.object({
      cardAccountId: z.string().uuid(),
      memberId: z.string().uuid().optional(),
      nickname: z.string().min(1).max(100).trim().optional(),
      last4: z
        .string()
        .length(4)
        .regex(/^\d{4}$/, "Must be 4 digits")
        .optional(),
    });
  }

  static readonly createSchema = this.schema;

  static override get bodySchema() {
    return super.bodySchema.extend(this.schema.shape);
  }
}
```

## Example Test

```typescript
import { assertEquals, assertThrows } from "@std/assert";
import { Card } from "@shared/entities/card.ts";

Deno.test("Card - creates with all optional fields omitted", () => {
  const data = {
    cardAccountId: crypto.randomUUID(),
    isActive: true,
  };

  const card = new Card(data);

  assertEquals(card.cardAccountId, data.cardAccountId);
  assertEquals(card.memberId, undefined);
  assertEquals(card.nickname, undefined);
  assertEquals(card.last4, undefined);
});

Deno.test("Card - creates with all fields", () => {
  const data = {
    cardAccountId: crypto.randomUUID(),
    memberId: crypto.randomUUID(),
    nickname: "Andrew's Main Card",
    last4: "1234",
    isActive: true,
  };

  const card = new Card(data);

  assertEquals(card.cardAccountId, data.cardAccountId);
  assertEquals(card.memberId, data.memberId);
  assertEquals(card.nickname, "Andrew's Main Card");
  assertEquals(card.last4, "1234");
});

Deno.test("Card - validates last4 format", () => {
  const baseData = {
    cardAccountId: crypto.randomUUID(),
    isActive: true,
  };

  // Valid
  const validCard = new Card({ ...baseData, last4: "9876" });
  assertEquals(validCard.last4, "9876");

  // Invalid - not 4 digits
  assertThrows(() => new Card({ ...baseData, last4: "12" }));
  assertThrows(() => new Card({ ...baseData, last4: "12345" }));
  assertThrows(() => new Card({ ...baseData, last4: "abcd" }));
});

Deno.test("Card - displayName formats correctly", () => {
  const cardAccountId = crypto.randomUUID();

  // With nickname and last4
  const card1 = new Card({
    cardAccountId,
    nickname: "Main Card",
    last4: "1234",
    isActive: true,
  });
  assertEquals(card1.displayName, "Main Card (••1234)");

  // With nickname only
  const card2 = new Card({
    cardAccountId,
    nickname: "Main Card",
    isActive: true,
  });
  assertEquals(card2.displayName, "Main Card");

  // With last4 only
  const card3 = new Card({
    cardAccountId,
    last4: "1234",
    isActive: true,
  });
  assertEquals(card3.displayName, "Card ••1234");

  // With neither
  const card4 = new Card({
    cardAccountId,
    isActive: true,
  });
  assertEquals(card4.displayName, "Card");
});

Deno.test("Card - trims nickname", () => {
  const card = new Card({
    cardAccountId: crypto.randomUUID(),
    nickname: "  Main Card  ",
    isActive: true,
  });

  assertEquals(card.nickname, "Main Card");
});

Deno.test("Card - validates UUID fields", () => {
  const baseData = {

  // Invalid cardAccountId
  assertThrows(() => new Card({ ...baseData, cardAccountId: "invalid-uuid" }));

  // Invalid memberId
  assertThrows(() => new Card({ ...baseData, memberId: "invalid-uuid" }));
});
```
