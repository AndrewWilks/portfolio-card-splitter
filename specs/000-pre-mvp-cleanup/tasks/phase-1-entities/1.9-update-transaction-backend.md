### TransactionService Updates

```typescript
import { CardAccountRepository } from "@backend/repositories";
import { CardRepository } from "@backend/repositories";

// Update constructor to inject card repositories
constructor(
  private transactionRepository: TransactionRepository,
  private merchantRepository: MerchantRepository,
  private tagRepository: TagRepository,
  private memberRepository: MemberRepository,
  private potRepository: PotRepository,
  private cardAccountRepository: CardAccountRepository, // NEW
  private cardRepository: CardRepository // NEW
) {}

// Update Zod schema
const CreateTransactionRequestSchema = z.object({
  transactionDate: z.string().refine((date) => !isNaN(Date.parse(date)), {
    message: "Invalid date format",
  }),
  amountCents: z.number().positive(),
  description: z.string().min(1).max(255),
  merchantId: z.string().uuid(),
  potId: z.string().uuid().optional(),
  memberId: z.string().uuid(),
  cardAccountId: z.string().uuid(), // NEW - REQUIRED
  cardId: z.string().uuid().optional(), // NEW
});

// Update createTransaction method
async createTransaction(request: unknown): Promise<Transaction> {
  const validated = CreateTransactionRequestSchema.parse(request);

  // Validate merchant exists
  const merchant = await this.merchantRepository.findById(
    validated.merchantId
  );
  if (!merchant) {
    throw new Error("Merchant not found");
  }

  // Validate member exists
  const member = await this.memberRepository.findById(validated.memberId);
  if (!member) {
    throw new Error("Member not found");
  }

  // NEW: Validate CardAccount exists
  const cardAccount = await this.cardAccountRepository.findById(
    validated.cardAccountId
  );
  if (!cardAccount) {
    throw new Error("CardAccount not found");
  }

  // NEW: If cardId provided, validate Card exists
  if (validated.cardId) {
    const card = await this.cardRepository.findById(validated.cardId);
    if (!card) {
      throw new Error("Card not found");
    }

    // Also validate Card belongs to CardAccount
    if (card.cardAccountId !== validated.cardAccountId) {
      throw new Error("Card does not belong to CardAccount");
    }
  }

  // If potId provided, validate pot exists
  if (validated.potId) {
    const pot = await this.potRepository.findById(validated.potId);
    if (!pot) {
      throw new Error("Pot not found");
    }
  }

  const transaction = new Transaction({
    transactionDate: new Date(validated.transactionDate),
    amountCents: validated.amountCents,
    description: validated.description,
    merchantId: validated.merchantId,
    potId: validated.potId,
    memberId: validated.memberId,
    cardAccountId: validated.cardAccountId, // NEW
    cardId: validated.cardId, // NEW
  });

  return await this.transactionRepository.save(transaction);
}
```

### Route Updates

**File**: `backend/routes/transactions/api_transactions_create.ts`

```typescript
// Request body now requires cardAccountId
export async function apiTransactionsCreate(
  c: Context,
  transactionService: TransactionService
) {
  try {
    const user = c.get("user");
    if (!user) {
      return c.json({ error: "Unauthorized" }, 401);
    }

    const body = await c.req.json();

    // Validate cardAccountId is provided
    if (!body.cardAccountId) {
      return c.json({ error: "cardAccountId is required" }, 400);
    }

    const transaction = await transactionService.createTransaction(body);

    return c.json(
      {
        transaction: {
          id: transaction.id,
          transactionDate: transaction.transactionDate.toISOString(),
          amountCents: transaction.amountCents,
          description: transaction.description,
          merchantId: transaction.merchantId,
          potId: transaction.potId,
          memberId: transaction.memberId,
          cardAccountId: transaction.cardAccountId, // NEW
          cardId: transaction.cardId, // NEW
          isActive: transaction.isActive,
          createdAt: transaction.createdAt.toISOString(),
          updatedAt: transaction.updatedAt.toISOString(),
        },
        message: "Transaction created successfully",
      },
      201
    );
  } catch (error) {
    if (error instanceof Error) {
      return c.json({ error: error.message }, 400);
    }
    return c.json({ error: "Internal server error" }, 500);
  }
}
```
