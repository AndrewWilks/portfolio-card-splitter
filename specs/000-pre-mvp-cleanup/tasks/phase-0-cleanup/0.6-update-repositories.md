### Reservation Repository

```typescript
async create(data: {
  potId: string;
  transactionId: string;
  allocationId?: string;  // Added
  memberId: string;       // Added
  amountCents: number;
}): Promise<Reservation> {
  const result = await this.db
    .insert(reservations)
    .values({
      potId: data.potId,
      transactionId: data.transactionId,
      allocationId: data.allocationId,  // Include
      memberId: data.memberId,          // Include
      amountCents: data.amountCents,
      createdById: data.createdById,
    })
    .returning();

  return Reservation.create(result[0]);
}

async findByMemberId(memberId: string): Promise<Reservation[]> {
  const results = await this.db
    .select()
    .from(reservations)
    .where(eq(reservations.memberId, memberId));

  return results.map(r => Reservation.create(r));
}
```
