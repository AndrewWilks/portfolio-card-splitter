## Example Updates

### Payment Service

```typescript
import { Payment } from "@shared/entities";
import {
  PaymentRepository,
  TransactionRepository,
  PotRepository,
} from "@backend/repositories";
import { Cents } from "@shared/types";

export class PaymentService {
  constructor(
    private paymentRepo: PaymentRepository,
    private transactionRepo: TransactionRepository,
    private potRepo: PotRepository
  ) {}

  async createPayment(data: {
    potId: string;
    transactionId: string;
    amountCents: Cents;
    paidOn: Date;
    reservationId?: string;
    note?: string;
  }): Promise<Payment> {
    // Validate pot exists
    const pot = await this.potRepo.findById(data.potId);
    if (!pot) {
      throw new Error("Pot not found");
    }

    // Validate transaction exists
    const transaction = await this.transactionRepo.findById(data.transactionId);
    if (!transaction) {
      throw new Error("Transaction not found");
    }

    // Validate payment doesn't exceed transaction total
    const existingPayments = await this.paymentRepo.findByTransactionId(
      data.transactionId
    );
    const totalPaid = existingPayments.reduce(
      (sum, p) => sum + p.amountCents,
      0
    );

    if (totalPaid + data.amountCents > transaction.amountCents) {
      throw new Error("Payment would exceed transaction total");
    }

    // Create payment entity
    const payment = Payment.create({
      potId: data.potId,
      transactionId: data.transactionId,
      amountCents: data.amountCents,
      paidOn: data.paidOn,
      reservationId: data.reservationId,
      note: data.note,
    });

    // Save and return
    const saved = await this.paymentRepo.save(payment);
    return saved[0] as Payment;
  }
}
```

### Payment Route

```typescript
import { Context } from "hono";
import { z } from "zod";
import { PaymentService } from "@backend/services";
import { zValidator } from "@hono/zod-validator";

const createPaymentSchema = z.object({
  potId: z.string().uuid(),
  transactionId: z.string().uuid(),
  amountCents: z.number().int().positive(),
  paidOn: z.string().datetime(), // ISO date string
  reservationId: z.string().uuid().optional(),
  note: z.string().optional(),
});

export function apiPaymentsCreate(c: Context, paymentService: PaymentService) {
  return async (c: Context) => {
    try {
      const body = await c.req.json();
      const validated = createPaymentSchema.parse(body);

      const payment = await paymentService.createPayment({
        ...validated,
        paidOn: new Date(validated.paidOn),
      });

      return c.json({ payment: payment.toJSON }, 201);
    } catch (error) {
      if (error instanceof z.ZodError) {
        return c.json(
          { error: "Validation failed", details: error.errors },
          400
        );
      }
      if (error instanceof Error) {
        return c.json({ error: error.message }, 400);
      }
      return c.json({ error: "Internal server error" }, 500);
    }
  };
}
```

### Reservation Service

```typescript
import { Reservation } from "@shared/entities";
import {
  ReservationRepository,
  PotRepository,
  TransactionRepository,
  MemberRepository,
  AllocationRepository,
} from "@backend/repositories";
import { Cents } from "@shared/types";

export class ReservationService {
  constructor(
    private reservationRepo: ReservationRepository,
    private potRepo: PotRepository,
    private transactionRepo: TransactionRepository,
    private memberRepo: MemberRepository,
    private allocationRepo: AllocationRepository
  ) {}

  async createReservation(data: {
    potId: string;
    transactionId: string;
    memberId: string;
    amountCents: Cents;
    allocationId?: string;
  }): Promise<Reservation> {
    // Validate entities exist
    const [pot, transaction, member] = await Promise.all([
      this.potRepo.findById(data.potId),
      this.transactionRepo.findById(data.transactionId),
      this.memberRepo.findById(data.memberId),
    ]);

    if (!pot) throw new Error("Pot not found");
    if (!transaction) throw new Error("Transaction not found");
    if (!member) throw new Error("Member not found");

    // Validate allocation if provided
    if (data.allocationId) {
      const allocation = await this.allocationRepo.findById(data.allocationId);
      if (!allocation) {
        throw new Error("Allocation not found");
      }
      // Could add validation: reservation amount <= allocation amount
    }

    // Create reservation entity
    const reservation = Reservation.create({
      potId: data.potId,
      transactionId: data.transactionId,
      memberId: data.memberId,
      amountCents: data.amountCents,
      allocationId: data.allocationId,
    });

    // Save and return
    const saved = await this.reservationRepo.save(reservation);
    return saved[0] as Reservation;
  }

  async deleteReservation(id: string): Promise<void> {
    // Hard delete (Reservation has no isActive field)
    await this.reservationRepo.delete(id);
  }
}
```
